## 2. Design (bukosabino/ta patterns adopted)

Upstream: `https://github.com/bukosabino/ta` (`pip install ta`, verified `0.11.0` in venv). Adopted canonical pattern — stateless indicator classes with `fillna=False`, then explicit NaN hygiene + single terminal shift in OUR code (never rely on `fillna=True`, which would inject forward-fill artifacts):

```python
import ta  # lazy import inside functions; ImportError -> fail-closed raise, never silent fallback

from ta.trend import SMAIndicator, EMAIndicator, MACD, ADXIndicator, CCIIndicator, AroonIndicator, PSARIndicator
from ta.momentum import RSIIndicator, StochasticOscillator, StochRSIIndicator, WilliamsRIndicator, ROCIndicator, TSIIndicator
from ta.volatility import BollingerBands, AverageTrueRange, DonchianChannel, KeltnerChannel
from ta.volume import OnBalanceVolumeIndicator, ChaikinMoneyFlowIndicator, MFIIndicator, ForceIndexIndicator

rsi2 = RSIIndicator(close=c, window=2, fillna=False).rsi()          # RSI-2
adx = ADXIndicator(high=h, low=l, close=c, window=14, fillna=False) # adx/pos/neg
bb = BollingerBands(close=c, window=20, window_dev=2, fillna=False) # hband/lband/mavg/pband/width
dc = DonchianChannel(high=h, low=l, close=c, window=20, fillna=False)
```

Rules for this repo (fail-closed, no-lookahead):

1. Input convention = gplearn's: lowercase `close/high/low/volume` columns (NOT `src/alpha/indicators.py` Capitalized convention). Missing `high/low` → trend/volatility families that need them yield NaN columns (still shifted); missing `volume` → volume family yields NaN columns. Never synthesize prices.
2. ONE terminal `.shift(1)` at the end of `build_ta_terminals` (same as `gplearn_factors.py:58`), plus qlib-style per-column shift equivalence. No per-indicator shifting (double-shift bug guard: assert in tests that `T["ta_rsi2"].iloc[2]` equals raw row-1 RSI-2).
3. `fillna=False` everywhere in `ta` calls; sanitize with `replace([inf, -inf], nan)` and leave NaN for warmup rows — `evolve()` already `.dropna()`s, `get_feature_matrix(dropna=True)` matches qlib. Degenerate series → NaN → dropped, never 0-filled into a fake signal (differs from `evaluate_program`'s 0-fill, which applies to *program output*, not terminals).
4. `ta_`-prefix on ALL 40+ columns (no collision with qlib `KMid_*`/`ROC_*` or gplearn `ret_1`). Ratios centered where gplearn does (`rsi` centered to [-1,1]-ish) for GP-friendliness.
5. gplearn index stability: existing `TERMINALS` order untouched. New names live in separate `TA_TERMINALS` tuple; `build_terminals_ta()` returns the concat `[list(TERMINALS) + list(TA_TERMINALS)]` with the same single `shift(1)`. `program_to_pandas`/`_parse` are NOT modified in this proposal (Xk space unchanged) — a follow-up edge-gate task can extend the parser once a TA terminal proves OOS value.
6. Offline-first: `ta` is pure pandas/numpy vectorized (no network). Tests run on synthetic OHLCV, zero network.
7. No change to `evolve_real.py` or `strategies/registry.json` (edge-gate compliance).

### Diff 1 — NEW `src/signals/ta_features.py` (the whole adoption; 40 features, past-only)

New file. Lowercase-bars in, shifted `ta_*` frame out; mirrors `compute_alpha158`'s registry/matrix API so `scripts/` and tests treat it identically.

```python
"""bukosabino/ta vectorized indicator features (past-only structured alpha).

Source: https://github.com/bukosabino/ta (verified ta==0.11.0 in venv).

Pipeline: lowercase OHLCV bars -> ta indicator classes (fillna=False) ->
centered/ratio normalization -> SINGLE .shift(1) (no lookahead, same as
gplearn_factors.build_terminals) -> ta_-prefixed terminal frame.

40 features across 4 families:
  trend (14): donchian hi/lo/width/pos x2 windows, ADX + DI+/DI-,
    macd line/signal/hist, SMA50 ratio, EMA20 ratio
  momentum (12): RSI-2, RSI-14 centered, Stoch k/d, StochRSI k/d,
    Williams %R, ROC-10, ROC-30, TSI, UO
  volatility (8): Bollinger hi/lo/width/pct x2 windows (20 + 10)
  volume (6): OBV slope, CMF-20, MFI-14, ForceIndex, VPT slope, NVI slope

Paper only. No orders placed; output is a feature frame. Research-only
until walk-forward + permutation + DSR pass (no registry.json entry).
"""
from __future__ import annotations

import numpy as np
import pandas as pd

TA_FEATURE_NAMES: list[str] = []

TREND_WINDOWS = (10, 20)
BB_WINDOWS = (10, 20)


def _register(name: str) -> str:
    TA_FEATURE_NAMES.append(name)
    return name


def _require_ta():
    try:
        import ta  # noqa: F401
    except ImportError as e:
        raise ImportError("ta==0.11.0 required (see requirements.txt); refusing silent fallback") from e


def build_ta_terminals(bars: pd.DataFrame) -> pd.DataFrame:
    """Past-only ta features; every column shifted by 1 (no lookahead)."""
    _require_ta()
    from ta.momentum import (
        ROCIndicator, RSIIndicator, StochasticOscillator, StochRSIIndicator,
        TSIIndicator, WilliamsRIndicator,
    )
    from ta.trend import ADXIndicator, EMAIndicator, MACD, SMAIndicator
    from ta.volatility import AverageTrueRange, BollingerBands, DonchianChannel
    from ta.volume import (
        ChaikinMoneyFlowIndicator, ForceIndexIndicator, MFIIndicator,
        NegativeVolumeIndexIndicator, OnBalanceVolumeIndicator,
        VolumePriceTrendIndicator,
    )

    global TA_FEATURE_NAMES
    TA_FEATURE_NAMES = []
    c = bars["close"].astype(float)
    out = pd.DataFrame(index=bars.index)
    has_hl = {"high", "low"}.issubset(bars.columns)
    h = bars["high"].astype(float) if has_hl else c
    lo = bars["low"].astype(float) if has_hl else c
    has_vol = "volume" in bars
    v = bars["volume"].astype(float) if has_vol else pd.Series(1.0, index=bars.index)

    # ---- TREND (14) ----
    for w in TREND_WINDOWS:
        dc = DonchianChannel(high=h, low=lo, close=c, window=w, fillna=False)
        out[_register(f"ta_dc_hi_{w}")] = dc.donchian_channel_hband() / c - 1
        out[_register(f"ta_dc_lo_{w}")] = c / dc.donchian_channel_lband() - 1
        out[_register(f"ta_dc_width_{w}")] = dc.donchian_channel_width()
        out[_register(f"ta_dc_pos_{w}")] = dc.donchian_channel_pband()
    adx = ADXIndicator(high=h, low=lo, close=c, window=14, fillna=False)
    out[_register("ta_adx_14")] = adx.adx() / 100
    out[_register("ta_di_pos_14")] = adx.adx_pos() / 100
    out[_register("ta_di_neg_14")] = adx.adx_neg() / 100
    macd = MACD(close=c, window_slow=26, window_fast=12, window_sign=9, fillna=False)
    out[_register("ta_macd")] = macd.macd() / c
    out[_register("ta_macd_signal")] = macd.macd_signal() / c
    out[_register("ta_macd_hist")] = macd.macd_diff() / c
    out[_register("ta_sma50_ratio")] = c / SMAIndicator(close=c, window=50, fillna=False).sma_indicator() - 1
    out[_register("ta_ema20_ratio")] = c / EMAIndicator(close=c, window=20, fillna=False).ema_indicator() - 1

    # ---- MOMENTUM (12) ----
    out[_register("ta_rsi2")] = (RSIIndicator(close=c, window=2, fillna=False).rsi() - 50) / 50
    out[_register("ta_rsi14")] = (RSIIndicator(close=c, window=14, fillna=False).rsi() - 50) / 50
    stoch = StochasticOscillator(high=h, low=lo, close=c, window=14, smooth_window=3, fillna=False)
    out[_register("ta_stoch_k")] = stoch.stoch() / 100
    out[_register("ta_stoch_d")] = stoch.stoch_signal() / 100
    srsi = StochRSIIndicator(close=c, window=14, smooth1=3, smooth2=3, fillna=False)
    out[_register("ta_stochrsi_k")] = srsi.stochrsi_k()
    out[_register("ta_stochrsi_d")] = srsi.stochrsi_d()
    out[_register("ta_willr_14")] = WilliamsRIndicator(high=h, low=lo, close=c, lbp=14, fillna=False).williams_r() / 100
    out[_register("ta_roc_10")] = ROCIndicator(close=c, window=10, fillna=False).roc()
    out[_register("ta_roc_30")] = ROCIndicator(close=c, window=30, fillna=False).roc()
    out[_register("ta_tsi")] = TSIIndicator(close=c, window_slow=25, window_fast=13, fillna=False).tsi() / 100
    from ta.momentum import UltimateOscillator
    out[_register("ta_uo")] = UltimateOscillator(high=h, low=lo, close=c, fillna=False).ultimate_oscillator() / 100

    # ---- VOLATILITY (8) ----
    for w in BB_WINDOWS:
        bb = BollingerBands(close=c, window=w, window_dev=2, fillna=False)
        out[_register(f"ta_bb_hi_{w}")] = bb.bollinger_hband() / c - 1
        out[_register(f"ta_bb_lo_{w}")] = c / bb.bollinger_lband() - 1
        out[_register(f"ta_bb_width_{w}")] = bb.bollinger_wband()
        out[_register(f"ta_bb_pct_{w}")] = bb.bollinger_pband()
    _ = AverageTrueRange  # reserved: ATR family already covered by src/alpha/indicators.py + qlib PriceRange_20

    # ---- VOLUME (6; NaN when volume absent) ----
    if has_vol:
        out[_register("ta_obv_slope_10")] = OnBalanceVolumeIndicator(close=c, volume=v, fillna=False).on_balance_volume().diff(10)
        out[_register("ta_cmf_20")] = ChaikinMoneyFlowIndicator(high=h, low=lo, close=c, volume=v, window=20, fillna=False).chaikin_money_flow()
        out[_register("ta_mfi_14")] = (MFIIndicator(high=h, low=lo, close=c, volume=v, window=14, fillna=False).money_flow_index() - 50) / 50
        out[_register("ta_fi")] = ForceIndexIndicator(close=c, volume=v, window=13, fillna=False).force_index()
        out[_register("ta_vpt_slope_10")] = VolumePriceTrendIndicator(close=c, volume=v, fillna=False).volume_price_trend().diff(10)
        out[_register("ta_nvi_slope_10")] = NegativeVolumeIndexIndicator(close=c, volume=v, fillna=False).negative_volume_index().diff(10)
    else:
        for name in ("ta_obv_slope_10", "ta_cmf_20", "ta_mfi_14", "ta_fi", "ta_vpt_slope_10", "ta_nvi_slope_10"):
            out[_register(name)] = np.nan

    out = out.replace([np.inf, -np.inf], np.nan)
    return out.shift(1)


def get_feature_names() -> list[str]:
    """Return the ta_ feature column names from the last build_ta_terminals call."""
    return list(TA_FEATURE_NAMES)


def get_feature_matrix(df: pd.DataFrame, features: list[str] | None = None,
                       dropna: bool = True) -> pd.DataFrame:
    """Return just the ta_ feature columns as a clean matrix (mirrors qlib)."""
    if features is None:
        features = [col for col in df.columns if col.startswith("ta_")]
    mat = df[features].copy()
    if dropna:
        mat = mat.dropna()
    return mat
```

```python
"""Tests for ta_features port: count, no-lookahead, determinism, offline."""
import numpy as np
import pandas as pd

from src.signals.ta_features import build_ta_terminals, get_feature_matrix, get_feature_names


def _bars(n=200):
    rng = np.random.RandomState(42)
    idx = pd.date_range("2020-01-01", periods=n, freq="D")
    close = pd.Series(100 + np.cumsum(rng.randn(n) * 0.5) + 20, index=idx)
    return pd.DataFrame({"close": close, "high": close + abs(rng.randn(n)),
                         "low": close - abs(rng.randn(n)),
                         "volume": rng.randint(1_000_000, 10_000_000, n).astype(float)}, index=idx)


def test_at_least_40_features():
    T = build_ta_terminals(_bars())
    assert len(get_feature_names()) >= 40


def test_no_lookahead_single_shift():
    bars = _bars()
    T = build_ta_terminals(bars)
    assert pd.isna(T.iloc[0]).all(axis=None) or T.iloc[0].isna().all()
    # row 2 holds raw row-1 information only (past-only)
    from ta.momentum import RSIIndicator
    raw = (RSIIndicator(close=bars["close"], window=2, fillna=False).rsi() - 50) / 50
    got, exp = T["ta_rsi2"].iloc[2], raw.iloc[1]
    assert (got == exp) or (pd.isna(got) and pd.isna(exp))


def test_families_present():
    names = get_feature_names()
    for prefix in ("ta_dc_", "ta_adx_14", "ta_macd", "ta_rsi", "ta_bb_", "ta_cmf_20", "ta_mfi_14"):
        assert any(n.startswith(prefix) for n in names), prefix


def test_determinism_and_matrix():
    T1, T2 = build_ta_terminals(_bars()), build_ta_terminals(_bars())
    pd.testing.assert_frame_equal(T1, T2)
    mat = get_feature_matrix(T1)
    assert len(mat) > 0 and not mat.isna().any(axis=None)


def test_missing_volume_yields_nan_not_crash():
    bars = _bars().drop(columns=["volume"])
    T = build_ta_terminals(bars)
    assert T["ta_cmf_20"].isna().all() and T["ta_adx_14"].dropna().shape[0] > 0
```
